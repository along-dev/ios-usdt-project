```sql
  FROM sys_base_menus

    OR component LIKE 'view/systemTools/%'
 ORDER BY parent_id, sort;
```

### 3.2 备份（**必做**）

```sql
CREATE TABLE sys_base_menus_bak_r08 AS SELECT * FROM sys_base_menus;
```

### 3.3 清理（**事务**）

```sql
START TRANSACTION;

-- ① 先删【子菜单】（parent_id 指向被删目录的）
DELETE FROM sys_base_menus
 WHERE component LIKE 'view/example/%'
    OR component LIKE 'view/systemTools/%';

-- ② 再删可能残留的【父目录】记录（若其 path 为 example / systemTools）
DELETE FROM sys_base_menus
 WHERE path IN ('example', 'systemTools')
   AND component LIKE 'view/%';

COMMIT;

-- 核查：应为 0
SELECT COUNT(*) AS remaining
  FROM sys_base_menus
 WHERE component LIKE 'view/example/%'
    OR component LIKE 'view/systemTools/%';
```
